% uk/book.tex -- the master file for the Ukrainian edition.
% Hand-edited. Build with: finishing/tools/build_edition.sh uk
%
% The edition shares the English preamble. \booklanguage and \bookmainfont,
% defined before it is input, set the main language and the face. P052 is
% URW's Palladio, the face the English book's Pagella was made from, in the
% version that carries Cyrillic; it is a system font here (urw-base35), not part
% of TeX Live, and it has no small caps. The build runs from manuscript/, like
% the English one, so every \input path below is relative to manuscript/.
%
% uk/sections.tex and uk/status.tex are GENERATED by finishing/tools/editions.py
% from sections/ORDER.tsv and uk/translations.tsv: a current translation comes
% from uk/sections/; an untranslated or stale section is the English file, set
% as English, and status.tex names the stale ones for the title page.
%
% The title, subtitle and notice below are drafts, proposed in
% uk/glossary.tsv and not yet approved or reviewed by a native reader.
\documentclass[11pt,oneside]{book}
\def\booklanguage{ukrainian}
\def\booklocale{ukrainian}
\def\bookmainfont{P052}
\input{uk/status}

% Untranslated sections are set under \selectlanguage{english}, for English
% hyphenation; their chapter heads, the contents and the bibliography still
% take the edition's Ukrainian captions.
\AtBeginDocument{\let\captionsenglish\captionsukrainian}
% The bibliography's note fields are English prose, some of it a paragraph
% long, and the preamble prints them at the end of every full citation (it
% saves each into \af@note, D-773; the references carry none). Until they are
% translated the Ukrainian edition drops the saved note, so its footnotes
% carry the citation alone.
\makeatletter\AtEveryCitekey{\undef\af@note}\makeatother
