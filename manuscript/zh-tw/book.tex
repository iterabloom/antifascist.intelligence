% zh-tw/book.tex -- the master file for the Traditional Chinese (Taiwan) edition.
% Hand-edited. Build with: finishing/tools/build_edition.sh zh-tw
%
% The edition shares the English preamble. Chinese characters are set by
% luatexja in Noto Serif CJK TC, with the zh_TW font metric from chinese-jfm,
% which centres the full-width comma and stop as Taiwan's typesetting does.
% Latin text keeps the English book's face (no \bookmainfont). No babel
% locale is loaded (no \booklocale): babel's chinese-traditional squeezes
% luatexja's punctuation and has no chapter name, so translated sections are
% set under English, which only governs Latin hyphenation here, and the
% captions below are the edition's own. The build runs from manuscript/, so every \input path
% below is relative to manuscript/.
%
% zh-tw/sections.tex and zh-tw/status.tex are GENERATED by
% finishing/tools/editions.py. biblatex has no Chinese strings; the edition's
% own are in zh-tw/biblatex-zh-tw.lbx.
%
% The title, subtitle and notice below are drafts, proposed in
% zh-tw/glossary.tsv and not yet approved or reviewed by a native reader.
\documentclass[11pt,oneside]{book}
\def\booklanguage{english}

\input{preamble}
% luatexja's default faces are Japanese ones not installed here; name the
% Noto CJK TC faces before it loads, or it stops on a missing font.
\makeatletter
\def\ltj@stdmcfont{name:NotoSerifCJKtc-Regular}
\def\ltj@stdgtfont{name:NotoSansCJKtc-Regular}
\makeatother
\usepackage{luatexja-fontspec}
% JFM=zh_TW centres the full-width comma and stop, as Taiwan sets them, and
% expects the font's Traditional Chinese glyphs, which Language= selects.
% Without it the comma and stop print squeezed against the next character.
\setmainjfont{Noto Serif CJK TC}[JFM=zh_TW, Language=Chinese Traditional,
  BoldFont=Noto Serif CJK TC Bold, ItalicFont=Noto Serif CJK TC,
  BoldItalicFont=Noto Serif CJK TC Bold]
% Captions and headings: 目錄; 第1章 for a chapter, 附錄A for an appendix,
% 第一部 for a part. The book class prints \@chapapp and the number, and the
% part's name and number; both are replaced.
\addto\captionsenglish{\renewcommand{\contentsname}{目錄}%
  \renewcommand{\appendixname}{附錄}\renewcommand{\bibname}{參考文獻}}
\makeatletter
% After \appendix the class sets \@chapapp to expand to \appendixname.
\def\afzhappapp{\appendixname}
\newcommand{\afzhchapnum}{\ifx\@chapapp\afzhappapp 附錄\thechapter
  \else 第\thechapter 章\fi}
\patchcmd{\@makechapterhead}{\@chapapp\space \thechapter}{\afzhchapnum}{}%
  {\PackageError{zh-tw}{chapter head not patched}{}}
\newcommand{\afzhpartnum}{第\ifcase\value{part}\or 一\or 二\or 三\or 四\or
  五\or 六\or 七\or 八\or 九\else\arabic{part}\fi 部}
% hyperref wraps \@part as \H@old@part, and nameref wraps that in turn,
% keeping the class's own as \NR@part.
\patchcmd{\NR@part}{\partname\nobreakspace\thepart}{\afzhpartnum}{}%
  {\PackageError{zh-tw}{part head not patched}{}}
\makeatother
% Every citation string in Chinese, in translated and English sections alike.
\DeclareLanguageMapping{english}{zh-tw/biblatex-zh-tw}
% As in every edition: the bibliography's English note fields are left out of
% the footnotes until they are translated.
\makeatletter\AtEveryCitekey{\undef\af@note}\makeatother
